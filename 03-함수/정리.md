3. 매개변수와 인자
function greet(name,age){ //매개변수 (params) - 받는 쪽 이름표
    return `${name} ${age}살`
}
greet("노벨",21) //argument (인자) - 실제 넣는값
greet("노벨") //"노벨 undefined살" <- 부족하면 undefined
greet("노벨",21,"서울") //"노벨 21살" <- 남으면 그냥 같다 버림

JS는 기본값이 있든 없든 개수가 안맞아도 에러가 안뜸.
기본값의 역할은 에러 방지가 아니라, 값이 안 들어왔을 때 undefined 대신 쓸 값을 정해두는 것.

function greet(name = "손님",age = 0){
    return `${name} ${age}살`
}
greet() //"손님 0살"

4. return

function add(a,b){
    return a+b
    console.log("여기는 실행 안됨") // return 만나면 함수 즉시종료
}

return 이 없으면 undefined가 나옴.

function noReturn(){
    console.log("출력만 함.")
}
const r = noReturn()
console.log(r) //undefined

"출력"과 "반환"은 완전히 다른 일
┌────────────────┬────────────────────────────────────┐
│                │              하는 일               │
├────────────────┼────────────────────────────────────┤
│ console.log(x) │ 화면에 보여주기만 함. 값을 못 받음 │
├────────────────┼────────────────────────────────────┤
│ return x       │ 값을 돌려줌. 받아서 또 쓸 수 있음  │
└────────────────┴────────────────────────────────────┘

// 출력만 하는 함수 - 재사용 불가
function  getGrade(s) {console.log("A")}
const g =  getGrade(95)
console.log(`점수: ${g}`) //점수: undefined

// 값을 돌려주는 함수
function getGrade(s) {return "A"}
const g = getGrade(95)
console.log(`점수: ${g}`) //점수: A

5.early return ★

else를 안쓰고 조건에 걸리면 바로 나가는 방식입니다.

function getGrade(score){
    if(score >= 90){
        return "A"
    }else if(score >= 80){
        return "B"
    }else {
        return "C"
    }
}
// early return — 위와 완전히 동일
function getGrade(score) {
  if (score >= 90) return "A"
  if (score >= 80) return "B"
  return "C"
}

Q1  아래 코드가 왜 undefined가 나오는지 정리하시오.

function getGrade(score) {
  console.log("A")
}
const result = getGrade(95)
console.log(result)   // undefined

A: getGrade 안에 return이 없어서. return 없는 함수는 항상 undefined를 반환한다.
▎ console.log는 화면에 출력만 할 뿐 값을 반환하지 않는다. 출력 ≠ 반환.

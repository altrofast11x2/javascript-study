3. 매개변수와 인자
function greet(name,age){ //매개변수 (params) - 받는 쪽 이름표
    return `${name} ${age}살`
}
greet("노벨",21) //argument (인자) - 실제 넣는값
greet("노벨") //"노밸 undefinded살" <- 부족하면 undefinded
greet("노벨",21,"서울") //"노밸 21살" <- 남으면 그냥 같다 버림

그래서 기본값을 쓰는거임. 기본값을 쓰면 개수가 안맞아도 에러가 안뜸.

function greet(name = "손님",age = 0){
    return `${name} ${age}살`
}
greet() //"손님 0살"

return 이 없으면 undefinded가 나옴.

function noReturn(){
    console.log("출력만 함.")
}
const r = noReturn()
console.log(r) //undefinded

"출력"과 "반환"은 완전히 다른 일
┌────────────────┬────────────────────────────────────┐
│                │              하는 일               │
├────────────────┼────────────────────────────────────┤
│ console.log(x) │ 화면에 보여주기만 함. 값을 못 받음 │
├────────────────┼────────────────────────────────────┤
│ return x       │ 값을 돌려줌. 받아서 또 쓸 수 있음  │
└────────────────┴────────────────────────────────────┘

// 출력만 하는 함수 - 재사용 불가
function getGrage(s) {console.log("A")}
const g = getGrage(95)
console.log(`점수: ${g}`) //점수: undefinded

// 값을 돌려주는 함수
function getGrade(s) {return "A"}
const g = getGrade(95)
console.log(`점수: ${g}`) //점수: "A"

5.early return ★

else를 안쓰고 조건에 걸리면 바로 나가는 방식입니다.

function getGrade(score){
    if(score >= 90){
        return "A"
    }else if(score >= 80){
        return "B"
    }else {
        return "C"
    }
}
// early return — 위와 완전히 동일
function getGrade(score) {
  if (score >= 90) return "A"
  if (score >= 80) return "B"
  return "C"
}
